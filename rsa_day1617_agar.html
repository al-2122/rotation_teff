# ── 2. Fit Poisson to formally confirm overdispersion ─────────
# We fit Poisson first to formally test whether its assumption
# (variance = mean) holds. If DHARMa flags overdispersion
# (p < 0.05, ratio >> 1), we reject Poisson and move to NB.
# Random effects: seeds nested in plates nested in batches —
# ignoring this clustering would pseudoreplicate as plants on
# the same plate share physical and nutritional environment.
tips_tmb_pois <- glmmTMB(
  Number.of.Root.Tips ~ Genotype + (1 | Batch/Plate/Seed),

# Pairwise Fisher's exact tests with Bonferroni correction
# Simple, intuitive, handles zero failures fine

library(dplyr)

df <- data.frame(
  Species    = c("Addisie","Beten","Dabbi","Karadebi","Manyi","Tsedey"),
  Germinated = c(47, 40, 44, 40, 40, 41),
  Failed     = c(1,  0,  0,  2,  5,  3)
)

# Pairwise Fisher's exact tests
species_list <- df$Species
results <- data.frame()

for (i in 1:(nrow(df)-1)) {
  for (j in (i+1):nrow(df)) {
    mat <- matrix(c(df$Germinated[i], df$Failed[i],
                    df$Germinated[j], df$Failed[j]),
                  nrow = 2)
    test <- fisher.test(mat)
    results <- rbind(results, data.frame(
      Contrast   = paste(species_list[i], "vs", species_list[j]),
      OddsRatio  = round(test$estimate, 2),
      p_raw      = round(test$p.value, 4)
    ))
  }
}

# Bonferroni correction
results$p_bonferroni <- round(p.adjust(results$p_raw, method = "bonferroni"), 4)
results$Significant  <- results$p_bonferroni < 0.05
print(results)
